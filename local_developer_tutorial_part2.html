8. Google Drive & Salesforce Connectors

Setting up OAuth callbacks, credentials encryption, and background file synchronization.

1. OAuth Callback Controller apps/invoice-be/routers/connectors.py

Exchanges redirect codes for provider tokens, encrypting them via Fernet AES before DB write:

Pythonfrom fastapi import APIRouter, Depends, HTTPException, Query
from sqlmodel import Session
from cryptography.fernet import Fernet
from database import get_session
from config import get_settings
import requests

router = APIRouter()

def encrypt_token(token: str, key: str) -> str:
    f = Fernet(key.encode())
    return f.encrypt(token.encode()).decode()

@router.get("/callback/google")
def handle_google_callback(code: str, tenant_id: str, db: Session = Depends(get_session)):
    settings = get_settings()
    token_url = "https://oauth2.googleapis.com/token"
    payload = {
        "code": code,
        "client_id": settings.GOOGLE_CLIENT_ID,
        "client_secret": settings.GOOGLE_CLIENT_SECRET,
        "redirect_uri": settings.GOOGLE_REDIRECT_URI,
        "grant_type": "authorization_code"
    }
    res = requests.post(token_url, data=payload)
    if res.status_code != 200:
        raise HTTPException(status_code=400, detail="Token exchange failed")
    
    tokens = res.json()
    enc_access = encrypt_token(tokens["access_token"], settings.TOKEN_ENCRYPTION_KEY)
    
    # Save connection to DB
    from models import TenantConnection
    conn = TenantConnection(tenant_id=tenant_id, provider="google_drive", encrypted_access_token=enc_access)
    db.add(conn)
    db.commit()
    return {"status": "success"}

class QueryState(TypedDict):
    question: str
    chat_history: List[dict]
    retrieved_docs: List[str]
    answer: str

def retrieve_docs_node(state: QueryState):
    # Vector DB search using Chroma client
    return {"retrieved_docs": ["Invoice total for Acme is 120.00"]}

def answer_node(state: QueryState):
    # Generates response matching retrieved documents
    return {"answer": f"According to invoices, your Acme total is 120.00."}

builder = StateGraph(QueryState)
builder.add_node("retrieve", retrieve_docs_node)
builder.add_node("answer", answer_node)
builder.set_entry_point("retrieve")
builder.add_edge("retrieve", "answer")
builder.add_edge("answer", END)

memory = MemorySaver()
query_graph = builder.compile(checkpointer=memory)

1. Hybrid Ingestion Status Tracking Hook apps/invoice-fe/hooks/useInvoiceIngestion.ts 
TypeScriptimport { useState } from 'react';

export function useInvoiceIngestion() {
  const [statuses, setStatuses] = useState<Record<string, string>>({});

  const trackIngestion = (batchId: string, jobIds: string[]) => {
    if (jobIds.length <= 5) {
      // Mode A: Polling
      jobIds.forEach(jobId => {
        const interval = setInterval(async () => {
          const res = await fetch(`/api/v1/invoices/status/${jobId}`);
          const data = await res.json();
          setStatuses(prev => ({ ...prev, [jobId]: data.status }));
          if (data.status !== 'PROCESSING') clearInterval(interval);
        }, 2000);
      });
    } else {
      // Mode B: Server-Sent Events
      const eventSource = new EventSource(`/api/v1/invoices/stream/${batchId}`);
      eventSource.onmessage = (e) => {
        const data = JSON.parse(e.data);
        if (data.batch_complete) {
          eventSource.close();
        } else {
          setStatuses(prev => ({ ...prev, [data.invoice_id]: data.event }));
        }
      };
    }
  };

  return { statuses, trackIngestion };
}
